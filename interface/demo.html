Portal Web Externo (Demostración)

Portal del Empleado OESVICA

Esta página simula cualquier sitio web externo donde desees activar el chatbot. En la esquina inferior derecha verás el botón flotante del chatbot.

Código para activar en tu propia página:

<script type="text/javascript">
  (function(d, t) {
      var v = d.createElement(t), s = d.getElementsByTagName(t)[0];
      v.onload = function() {
        window.oesvicaChat.load({
          serverUrl: window.location.origin, // o la URL de tu servidor
          title: 'Atención al Trabajador OESVICA',
          position: 'right'
        });
      };
      v.src = '/interface/widget.js'; 
      v.type = 'text/javascript'; 
      s.parentNode.insertBefore(v, s);
  })(document, 'script');
</script>
